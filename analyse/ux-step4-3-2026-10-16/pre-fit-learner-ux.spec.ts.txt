test("Supabase auth startup preserves native links, callbacks and existing sessions", () => {
  const base = {
    isNativePlatform: false,
    pathname: "/",
    search: "",
    hash: "",
    storageKey: "sb-mcy-playwright-auth-token",
    storageKeys: [] as string[],
  };
  expect(shouldInitializeAuthOnLaunch(base)).toBe(false);
  expect(shouldInitializeAuthOnLaunch({ ...base, isNativePlatform: true })).toBe(true);
  expect(shouldInitializeAuthOnLaunch({ ...base, pathname: "/auth/callback" })).toBe(true);
  expect(shouldInitializeAuthOnLaunch({ ...base, search: "?code=pkce-code" })).toBe(true);
  expect(shouldInitializeAuthOnLaunch({ ...base, hash: "#access_token=legacy-session" })).toBe(true);
  expect(shouldInitializeAuthOnLaunch({ ...base, storageKeys: ["sb-mcy-playwright-auth-token"] })).toBe(true);
  expect(shouldInitializeAuthOnLaunch({ ...base, storageKeys: ["sb-another-project-auth-token"] })).toBe(false);
  expect(shouldInitializeAuthOnLaunch({ ...base, storageKeys: null })).toBe(true);
});

test("first-time guest defers the Supabase SDK until account authentication is requested", async ({ page }) => {
  const scriptRequests = new Set<string>();
  const authRequests: string[] = [];
  page.on("request", request => {
    if (request.resourceType() === "script") scriptRequests.add(request.url());
  });
  await page.route("https://mcy-playwright.invalid/**", async route => {
    const requestUrl = new URL(route.request().url());
    authRequests.push(`${route.request().method()} ${requestUrl.pathname}`);
    await route.fulfill({
      status: 200,
      contentType: "application/json",
      body: JSON.stringify({ message_id: "fictional-auth-message" }),
    });
  });

  await page.goto("/?mode=web");
  await expect(page.getByRole("button", { name: "Tạo tài khoản" })).toBeVisible();
  const initialScriptRequests = new Set(scriptRequests);
  await page.getByRole("button", { name: "Tạo tài khoản" }).click();
  await page.getByLabel("Địa chỉ email").fill("learner@example.test");
  await page.getByRole("button", { name: "Gửi email xác minh" }).click();
  await expect.poll(() => authRequests.some(request => request.endsWith("/auth/v1/otp"))).toBe(true);
  await expect.poll(() => [...scriptRequests].some(url => !initialScriptRequests.has(url))).toBe(true);
});

test("native auth accepts only a pending PKCE code on the exact app callback", () => {
  const now = 1_790_700_000_000;
  const expected = "motchutyoga://auth/callback";
  expect(nativeAuthCodeFromCallback(`${expected}?code=valid-code`, expected, String(now), now)).toBe("valid-code");
  expect(nativeAuthCodeFromCallback(`${expected}#access_token=attacker&refresh_token=token`, expected, String(now), now)).toBeNull();
  expect(nativeAuthCodeFromCallback("motchutyoga://other/callback?code=valid-code", expected, String(now), now)).toBeNull();
  expect(nativeAuthCodeFromCallback(`${expected}?code=valid-code`, expected, String(now - 16 * 60_000), now)).toBeNull();
  expect(nativeAuthCodeFromCallback(`${expected}?code=valid-code`, expected, null, now)).toBeNull();
});

test("native auth callback de-duplication rejects oversized URLs and bounds retained entries", () => {
  const handled = new Set<string>();
  expect(rememberNativeAuthUrl(handled, "")).toBe(false);
  expect(rememberNativeAuthUrl(handled, "x".repeat(MAX_NATIVE_AUTH_CALLBACK_URL_LENGTH + 1))).toBe(false);
  expect(handled.size).toBe(0);
  expect(rememberNativeAuthUrl(handled, "motchutyoga://auth/callback?code=first")).toBe(true);
  expect(rememberNativeAuthUrl(handled, "motchutyoga://auth/callback?code=first")).toBe(false);

  for (let index = 0; index < MAX_HANDLED_NATIVE_AUTH_URLS + 4; index += 1) {
    expect(rememberNativeAuthUrl(handled, `motchutyoga://auth/callback?code=${index}`)).toBe(true);
    expect(handled.size).toBeLessThanOrEqual(MAX_HANDLED_NATIVE_AUTH_URLS);
  }
  expect(handled.size).toBe(MAX_HANDLED_NATIVE_AUTH_URLS);
  expect(handled.has("motchutyoga://auth/callback?code=first")).toBe(false);
});

  const valid = await readBoundedJson(new Request("https://local.test", { method: "POST", body: JSON.stringify({ message: "slow pace" }) }), 64);
  expect(valid).toEqual({ message: "slow pace" });
  await expect(readBoundedJson(new Request("https://local.test", { method: "POST", body: "x".repeat(100) }), 32)).rejects.toBeInstanceOf(RequestBodyTooLargeError);

test("body-region contract keeps locale-neutral regions, sides and bilateral edits distinct", () => {
  expect(bodyRegionCodes("elbow", "left")).toEqual(["elbow_left"]);
  expect(bodyRegionCodes("knee", "both")).toEqual(["knee_left", "knee_right"]);
  expect(bodyRegionCodes("foot", "unspecified")).toEqual(["foot_unspecified"]);
  expect(decodeBodyRegionCode("forearm_wrist_right")).toEqual({ region: "forearm_wrist", side: "right" });
  expect(decodeBodyRegionCode("shoulder_left")).toEqual({ region: "shoulder", side: "left" });

  const left = { region: "elbow", side: "left", level: "Moderate" } as const;
  const right = { region: "elbow", side: "right", level: "Mild" } as const;
  const both = { region: "elbow", side: "both", level: "Mild" } as const;
  const paired = replaceBodyObservation([left], right, "right");
  expect(paired).toEqual([left, right]);
  expect(replaceBodyObservation(paired, both, "left")).toEqual([both]);
  expect(removeBodyObservation(paired, "elbow", "left")).toEqual([right]);
  expect(bodyRegionCodes("upper_back", "unspecified")).toEqual(["upper_back"]);
  expect(toCheckinObservationRpcPayload([{
    regionCode: "elbow_left",
    state: "discomfort",
    level: 2,
    note: "typing",
  }])).toEqual([{
    region_code: "elbow_left",
    state: "discomfort",
    level: 2,
    note: "typing",
  }]);
});

async function activate(target: Locator, touch: boolean) {
  if (touch) await target.tap();
  else await target.click();
}

  if (locale === "en-US") await expect(page.locator(".mcy-header-title")).toHaveText("A Little Yoga");
  await activate(page.locator(".mcy-body-disclosure"), touch);
}

  await expect(page.locator(".mcy-course-card")).toHaveCount(3);
  const catalogReadability = await page.evaluate(() => {
    const score = (selector: string) => {
      const element = document.querySelector(selector)!;
      const parse = (color: string) => color.match(/[\d.]+/g)!.slice(0, 3).map(Number);
      const luminance = (color: string) => {
        const [red, green, blue] = parse(color).map(channel => {
          const value = channel / 255;
          return value <= 0.04045 ? value / 12.92 : ((value + 0.055) / 1.055) ** 2.4;
        });
        return red * 0.2126 + green * 0.7152 + blue * 0.0722;
      };
      let node: Element | null = element;
      let surface = "rgba(0, 0, 0, 0)";
      while (node && (surface === "rgba(0, 0, 0, 0)" || surface === "transparent")) {
        surface = getComputedStyle(node).backgroundColor;
        if (surface === "rgba(0, 0, 0, 0)" || surface === "transparent") node = node.parentElement;
      }
      const foreground = luminance(getComputedStyle(element).color);
      const background = luminance(surface);
      return (Math.max(foreground, background) + 0.05) / (Math.min(foreground, background) + 0.05);
    };
    return {
      pill: Number.parseFloat(getComputedStyle(document.querySelector(".mcy-course-heading .mcy-state-pill")!).fontSize),
      metadata: Number.parseFloat(getComputedStyle(document.querySelector(".mcy-course-card small")!).fontSize),
      description: Number.parseFloat(getComputedStyle(document.querySelector(".mcy-course-card em")!).fontSize),
      contrasts: [
        score(".mcy-course-heading .mcy-state-pill"),
        score(".mcy-course-card small"),
        score(".mcy-course-card em"),
      ],
    };
  });
  expect(catalogReadability.pill).toBeGreaterThanOrEqual(11);
  expect(catalogReadability.metadata).toBeGreaterThanOrEqual(14);

  await page.getByRole("button", { name: /Một chút chuyển động nhẹ/ }).click();

  const detailReadability = await page.evaluate(() => {
    const rgb = (value: string) => value.match(/[\d.]+/g)!.slice(0, 3).map(Number);
    const luminance = (value: string) => {
      const channels = rgb(value).map(channel => {

        const color = getComputedStyle(node).backgroundColor;

      return getComputedStyle(document.body).backgroundColor;
    };

      const element = document.querySelector(selector)!;
      const style = getComputedStyle(element);
      const foreground = luminance(style.color);
      const surface = luminance(background(element));
      return {
        fontSize: Number.parseFloat(style.fontSize),
        contrast: (Math.max(foreground, surface) + 0.05) / (Math.min(foreground, surface) + 0.05),
      };
    };

      eyebrow: measure(".mcy-course-detail-page > .mcy-eyebrow"),

      lessonNumber: measure(".mcy-course-lesson-number"),
    };
  });

      const footerRect = footer.getBoundingClientRect();
      const actionRect = footer.querySelector(".mcy-action")!.getBoundingClientRect();
      return {
        footerBottom: footerRect.bottom,
        actionBottom: actionRect.bottom,
        viewportHeight: window.innerHeight,
        leftInset: actionRect.left,
        rightInset: window.innerWidth - actionRect.right,
      };
    });
    const initial = await metrics();
    expect(initial.footerBottom).toBeLessThanOrEqual(initial.viewportHeight);
    expect(initial.actionBottom).toBeLessThanOrEqual(initial.viewportHeight);
    expect(initial.leftInset).toBeGreaterThanOrEqual(15);
    expect(initial.rightInset).toBeGreaterThanOrEqual(15);

    await page.locator(".mcy-scroll").evaluate(scroll => scroll.scrollTo({ top: scroll.scrollHeight, behavior: "instant" }));
    await expect(action).toBeInViewport();
    const afterScroll = await metrics();
    expect(afterScroll.actionBottom).toBeLessThanOrEqual(afterScroll.viewportHeight);
    await action.click();
    await expect(page.getByRole("heading", { name: "Một chút thời gian cho chính mình." })).toBeVisible();
  }
});

test("bottom navigation has an opaque surface and readable muted labels", async ({ page }) => {
  await page.setViewportSize({ width: 390, height: 844 });
  await page.goto("/?mode=web");
  await page.getByRole("button", { name: "Thử tìm hiểu như khách" }).click();
  await page.getByRole("button", { name: "Tự chọn bài học miễn phí" }).click();

  const inactiveItems = page.locator(".mcy-bottom-nav button:not([aria-current='page'])");
  await expect(inactiveItems).toHaveCount(3);
  const styles = await inactiveItems.first().evaluate((button) => ({
    color: getComputedStyle(button).color,
    fontSize: getComputedStyle(button).fontSize,
    navigationBackground: getComputedStyle(button.closest(".mcy-bottom-nav")!).backgroundColor,
  }));
  expect(styles.color).toBe("rgb(86, 102, 90)");
  expect(Number.parseFloat(styles.fontSize)).toBeGreaterThanOrEqual(11);
  expect(styles.navigationBackground).toBe("rgb(255, 253, 249)");
});
